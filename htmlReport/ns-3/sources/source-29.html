 package tn.esprit.pi.domain;
 
 import jakarta.persistence.*;
 import lombok.*;
 import java.util.Set;
 
 @Entity
 @Getter
 @Setter
 @NoArgsConstructor
 @AllArgsConstructor
 public class VirtualTeam {
     @Id

     private SportType sportType;
     
     private Double earnedPoints;
     private Double weekPoints;
     
     @ManyToOne
